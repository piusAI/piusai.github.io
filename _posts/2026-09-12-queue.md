``` cpp
#pragma once
#include <iostream>
using namespace std;
#include <assert.h>

template <typename T>
class Vector {

public:

	explicit Vector(int capacity=0) :_buffer((capacity>0)? new T[capacity] : nullptr), _capacity(capacity) {}
	~Vector() {
		delete[] _buffer;
	}
	void Print() {
		for (int i = 0; i < _size; i++)
		{
			cout << _buffer[i] << endl;
		}
	}

	bool operator==(const Vector& other)
	{

		return _buffer == other._buffer && _size == other._size && _capacity == other._capacity;
	}
	Vector& operator= (const Vector& other) {
		if (*this == other) return *this;

		_capacity = other._capacity;
		_size = other._size;
		T* _newbuffer = new T[_capacity];
		for (int i = 0; i < _size; i++)
		{
			_newbuffer[i] = other._buffer[i];
		}
		_buffer = _newbuffer;
		return *this;
	}

	

	void Clear()
	{
		delete[] _buffer;
		_buffer = nullptr;
		_size = 0;
	}

	void pop_back()
	{
		T last = _buffer[_size];
		_size--;
			
	}

	T& back() { return _buffer[_size - 1]; }

	void push_back(T data) {
		if (_capacity == _size)
		{
			int newCapacity = int(_capacity * 1.5);
			if (newCapacity == 1) ++newCapacity;
			reserve(newCapacity);
		}
		_buffer[_size] = data;
		_size++;

	}
	void reserve(int newCapacity) {
		if (newCapacity == _capacity) return;

		T* _newBuffer = new T[newCapacity];

		for (int i = 0; i < _size; i++)
		{
			_newBuffer[i] = _buffer[i];
		}
		delete[] _buffer;
		_buffer = _newBuffer;
		_capacity = newCapacity;
	}
	void resize(int capacity)
	{
		reserve(capacity);
		_capacity = capacity;
		
	}

	T* begin() { return _buffer; }
	T* end() { return _buffer + _size; }

	T& operator[](int index)
	{
		assert(index >= 0 && index < _capacity);
		return _buffer[index];
	}

	bool empty() { return Size()<1; }
	int Size() { return _size; }
	int Capacity() { return _capacity; }
	void SetSize(int size) { _size = size; }

public:
	T* _buffer = nullptr;
	int _capacity = 0;
	int _size = 0;

};
```

``` cpp
			Vector<T> newContainer;

				newContainer[i] = _container[(_front+i) %oldCapacity];
			}
			_container = newContainer;
			_front = 0;
			_back = _size;
		}
		_container[_back] = data;
		_back = (_back + 1) % _container.Capacity();
		_size++;
	}

		_front = (_front + 1)% _container.Capacity();
```
