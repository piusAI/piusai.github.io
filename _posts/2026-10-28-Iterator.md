[Vector](https://piusai.github.io/c++/2026/09/16/Container.html)와 같은 Container에서 Traversal(순회)하기 위해서 많이 활용되는 Iterator,  
이번 글에서는 Iterator를 직접 구현해보겠다.

iterator를 지원하려면 각 클래스가 다음을 갖춰야한다.
- Iterator class : 
  - `operator*, operator++(), operator++(int), operator+(int c)`
  - 역참조, 전위, 후위, 건너뜀

- Container(Vector): `begin(), end()`
### iterator, 직접 구현

``` cpp
#include <iostream>
using namespace std;
#include <vector>
#include <list>
#include <assert.h>

template <typename T>
class Iterator {
public:
	Iterator():_ptr(nullptr) {}
	Iterator(T* ptr) : _ptr(ptr) {}

	T& operator *() {
		return *_ptr;
	}
	
	//전위
	Iterator& operator++() {
		_ptr++;
		return *this;
	}
	
	//후위, 전위랑 구분하기 위한 더미인자
	Iterator operator++(int) {
		Iterator temp = *this;
		++_ptr;
		return temp;
	}

	Iterator operator+(const int counter) const
	{
		Iterator temp = *this;
		temp._ptr += counter;
		return temp;
	}

	bool operator ==(const Iterator& other) const{ return _ptr == other._ptr; }

	bool operator !=(const Iterator& other) const {	return _ptr != other._ptr; }

public:
	T* _ptr;
};

template <typename T>
class Vector
{
public:

	explicit Vector() {}
	Vector(int capacity) : _capacity(capacity) {
		_buffer = new T[_capacity];
	}
	~Vector()
	{
		delete[] _buffer;
	}

	//복사 생성자
	Vector(const Vector& other):_capacity(other._capacity), _size(other._size)
	{
		_buffer = new T[_capacity];
		for (int i = 0; i < _size; i++)
		{
			_buffer[i] = other._buffer[i];
		}
	}
	

	using iterator = Iterator<T>;
	iterator begin() { return iterator(_buffer); } //Iterator(T*) 생성자로 간다
	iterator end() { return begin() + _size; }

	//깊은 복사
	Vector& operator=(const Vector& other){
		if (this == &other) return *this;
		delete[] _buffer;

		_capacity = other._capacity;
		_size = other._size;
		_buffer = new T[_capacity];
		for (int i = 0; i < _size; i++)
		{
			_buffer[i] = other._buffer[i];
		}
		return *this;
	}


	void clear()
	{
		_size = 0;
	}

	void push_back(const T& data)
	{
		if (_size == _capacity)
		{
			int newCapacity = static_cast<int>(_capacity * 1.5);
			if (newCapacity == _capacity) newCapacity++;
			reserve(newCapacity);
		}
		_buffer[_size] = data;
		_size++;
	}

	void pop_back()
	{
		_size--;
	}
	void reserve(int newCapacity)
	{
		if (newCapacity <= _capacity) return;
		else {
			_capacity = newCapacity;
		}
		T* newbuffer = new T[newCapacity];
		for (int i = 0; i < _size; i++)
			newbuffer[i] = _buffer[i];
		if (_buffer) delete[] _buffer;

		_buffer = newbuffer;
	}

	T& operator[](int index)
	{
		assert(index >= 0 && index < _size);
		return _buffer[index];
	}



	int size() { return _size; }
	int capacity() { return _capacity; }

private:
	T* _buffer = nullptr;
	int _capacity = 0;
	int _size = 0;
};

int main()
{
	Vector<int> v(2);
	v.push_back(10);
	v.push_back(20);
	v.push_back(30);
	v.push_back(40);

	Vector<int>::iterator it;


	for (it=v.begin(); it != v.end(); it++)
	{
		cout << *it <<endl;
		
	}
}
```


- iterator의 `_ptr`은 현재 가리키는 원소 위치를 들고 있다.
- Vector의 `_buffer`는 그 배열 시작주소를 들고있다. `new[]/delete[]`로 메모리 관리


- `begin()`으로 `Iterator(T*)` 생성자 호출할때, `_buffer`의 주소를 복사해서 `_ptr`에 넣기 때문에 같은 주소를 가리킴.  
  -> 시작 순간(`begin()`)에는 같은 주소를 가리킨다.
- 그 이후 iterator는 ++로 `_ptr`을 옮긴다. - 이때부터 `_buffer`는 그대로기에 달라짐
